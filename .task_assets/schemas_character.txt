    91	def _normalize_information_ids(model: _IdCarrier) -> _IdCarrier:
    92	    """把 `new_information_ids` 归一到 `information_ids`，两侧保持同一份内容。
    93	
    94	    PRD §7.1 要求兼容旧字段一个版本周期，但内部立即以 `information_ids` 为准。
    95	    双向同步而不是单向覆盖，才能让仍在读旧字段的调用方继续正确工作。
    96	    """
    97	    new_ids = [str(x).strip() for x in (model.new_information_ids or []) if str(x).strip()]
    98	    info_ids = [str(x).strip() for x in (model.information_ids or []) if str(x).strip()]
    99	    merged = list(dict.fromkeys([*info_ids, *new_ids]))
   100	    if merged != info_ids:
   101	        model.information_ids = merged
   102	    if merged != new_ids:
   103	        model.new_information_ids = merged
   104	    return model

   155	                                     # 代码回退为 evidence_chapter_index（见 valid_from_is_fallback）

   158	    # 回落标注（事故修复：状态事实回填 100% 拒绝一事排查出的相关修正，见

   166	    # 或模型申报的边界本身独立核验通过（值就是模型的原始申报值）。矛盾边界（如申报的终点早于
   167	    # 证据章）不属于本标注范围——那种情况下整条事实都不会被登记。
