4705: ) -> str:
4706:     """Deterministically retrieve a bounded source dossier for one character."""
4707:     anchors = [value.strip() for value in appellations if value and value.strip()]
4708:     selected: list[str] = []
4709:     seen: set[tuple[int, str]] = set()
4710:     for chapter in chapters:
4711:         content = (chapter.get("content") or "").strip()
4712:         if not content:
4713:             continue
4714:         try:
4715:             chapter_idx = int(chapter.get("idx"))
4716:         except (TypeError, ValueError):
4717:             continue
4718:         for segment in index_source_segments(content, max_chars=800):
4719:             text = segment.text.strip()
4720:             if not text or not any(anchor in text for anchor in anchors):
4721:                 continue
4722:             key = (chapter_idx, text)
4723:             if key in seen:
4724:                 continue
4725:             block = f"【第{chapter_idx}章·证据】\n{text}"
4726:             if sum(len(item) for item in selected) + len(block) > max_chars:
4727:                 return "\n\n".join(selected)
4728:             selected.append(block)
4729:             seen.add(key)
4730:             if len(selected) >= BIBLE_DETAIL_EVIDENCE_MAX_SEGMENTS:
4731:                 return "\n\n".join(selected)
4732:     if selected:
4733:         return "\n\n".join(selected)
4734:     # No lexical hit: bounded fallback only, never the 60K source corpus.
4735:     for chapter in chapters[:3]:
4736:         content = (chapter.get("content") or "").strip()

4738:             continue
4739:         block = f"【第{chapter.get('idx', '?')}章·有限背景】\n{content[:1200]}"
4740:         if sum(len(item) for item in selected) + len(block) > max_chars:
4741:             break
4742:         selected.append(block)
