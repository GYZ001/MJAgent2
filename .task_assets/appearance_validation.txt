  2511	APPEARANCE_EVIDENCE_QUOTE_MAX_CHARS = 40

  2514	def _appearance_evidence_verified(
  2515	    chapters_by_idx: dict[int, str],
  2516	    anchor_texts: set[str],
  2517	    evidence_chapter_index: int,
  2518	    evidence_quote: str,
  2519	) -> bool:
  2520	    """标志性特征证据核验：结构性判据，不做任何语义分类（禁止黑白名单式修复）。
  2521	
  2522	    两个条件必须同时成立：
  2523	    1. evidence_quote 长度 <= APPEARANCE_EVIDENCE_QUOTE_MAX_CHARS，且是
  2524	       evidence_chapter_index 对应章节原文的逐字连续子串（按 `_quote_comparison_variants`
  2525	       产出的候选引句形式判断，与别名核验同一套脱引号容错）；
  2526	    2. 角色规范名（或调用方传入的其它已确认锚点）出现在这条引句本身内部——不是出现在
  2527	       整章的其它位置。这是与 `_alias_declaration_verified` 条件 3（整章共现）的关键
  2528	       区别：外观证据要求"名字和描写在同一条不超过 40 字的短引句里"，因为整章共现挡不住
  2529	       "同一句里名字属于A、描写属于B"这种跨人借用（王有材事故的实际触发路径）。
  2530	    """
  2531	    quote = (evidence_quote or "").strip()
  2532	    if not quote or len(quote) > APPEARANCE_EVIDENCE_QUOTE_MAX_CHARS:
  2533	        return False
  2534	    chapter_text = chapters_by_idx.get(evidence_chapter_index, "")
  2535	    if not chapter_text:
  2536	        return False
  2537	    for candidate in _quote_comparison_variants(quote):
  2538	        if candidate in chapter_text and any(
  2539	            anchor and anchor in candidate for anchor in anchor_texts
  2540	        ):
  2541	            return True
  2542	    return False

  2545	def _validate_appearance_evidence(bible: Bible, chapters_by_idx: dict[int, str]) -> list[str]:
  2546	    """遍历每个角色的 source_evidence，只对非空条目核验；核验失败才产生 error（驱动
  2547	    AgentLoop 的修复重试）。空 source_evidence 数组永远不产生 error——诚实的"这个角色
  2548	    没有可举证的标志性特征"是安全默认值，不是缺陷信号，不能让"老实说没有"比"编一个能蒙混
  2549	    过关的"更差（那会复刻本次事故的激励结构）。
  2550	
  2551	    锚点只用角色规范名，不用 aliases：本函数在 AgentLoop 校验闭包里对候选 Bible 逐轮调用，
  2552	    此时 aliases 只是模型本轮申报、尚未经过 `_verify_character_aliases_in_place` 代码核验，
  2553	    用未核验的申报去解锁另一项核验会开一个"自证"漏洞——与 `_verify_character_aliases_in_place`
  2554	    自身"只用已验证别名扩大锚点集合"的既有规则一致（不确定不采信）。
  2555	    """
  2556	    errors: list[str] = []
  2557	    for i, character in enumerate(bible.characters):
  2558	        anchor_texts = {character.name}
  2559	        for j, evidence in enumerate(character.source_evidence):
  2560	            if _appearance_evidence_verified(
  2561	                chapters_by_idx, anchor_texts,
  2562	                evidence.evidence_chapter_index, evidence.evidence_quote,
  2563	            ):
  2564	                continue
  2565	            errors.append(
  2566	                f"characters[{i}]({character.name}).source_evidence[{j}] 未通过核验：第 "
  2567	                f"{evidence.evidence_chapter_index} 章原文里找不到一条不超过 "
  2568	                f"{APPEARANCE_EVIDENCE_QUOTE_MAX_CHARS} 字、且与「{character.name}」同句"
  2569	                "出现的逐字引句；请换一条真实可查的原文短引句，或直接去掉这个特征，"
  2570	                "appearance_canonical 只写通用形态即可，不必凑数。"
  2571	            )
  2572	    return errors

  2615	def _alias_bridge_quote(chapter_text: str, text: str) -> str | None:
  2616	    """从桥接章原文里确定性截取包含 text 的引句：复用 `index_source_segments`

  2619	    `_alias_declaration_verified` 条件 1（逐字子串）与条件 2（text 是引句子串）。

  2621	    交由调用方兜底拒绝，不强行拼一句可能不含 text 的引句。
  2622	
  2623	    不看主体——这是 `_find_alias_bridge_chapter` 优先找不到双锚定分段/双锚定章时的

  2625	    完全一致，不单独调用时不受影响。"""
  2626	    for segment in index_source_segments(chapter_text, max_chars=_ALIAS_BRIDGE_QUOTE_MAX_CHARS):
