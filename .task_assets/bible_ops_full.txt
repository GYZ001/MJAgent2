   444	async def _scene_bible_task(
   445	    project_id: str,
   446	    *,
   447	    parent_run_id: str | None = None,
   448	    requested_by: str = "system",
   449	    trigger_type: str = "automatic",
   450	) -> None:
   451	    """生成并落库场景清单，不绕过费用确认自动生成图片。"""
   452	    from app.stages import generate_scene_bible
   453	    conn = get_conn()
   454	    recorder = WorkflowRecorder.create(
   455	        workflow_type="scene_bible",
   456	        scope_type="project",
   457	        scope_id=project_id,
   458	        input_fingerprint=fingerprint(project_id, "scene_bible"),
   459	        requested_by=requested_by,
   460	        trigger_type=trigger_type,
   461	        policy_snapshot={"max_iterations": 4, "warning_blocks_downstream": True},
   462	        parent_run_id=parent_run_id,
   463	    )
   464	    try:
   465	        recorder.start()
   466	        p = conn.execute("SELECT bible_json FROM projects WHERE id=?", (project_id,)).fetchone()
   467	        if not p or not p["bible_json"]:
   468	            raise ValueError("人物谱不存在，不能生成场景 Bible")
   469	        bible = Bible.model_validate(json.loads(p["bible_json"]))
   470	        # 初始场景清单只取前 N 章：避免一上来就铺满全片场景；更靠后的新场景留到分镜阶段反应式补图。
   471	        from app.scenes import SCENE_BIBLE_CHAPTER_WINDOW
   472	        chapters = rows_to_dicts(conn.execute(
   473	            "SELECT * FROM chapters WHERE project_id=? ORDER BY idx LIMIT ?",
   474	            (project_id, SCENE_BIBLE_CHAPTER_WINDOW)).fetchall())
   475	        _, scenes = await recorder.step(
   476	            "scene_bible",
   477	            lambda: generate_scene_bible(chapters, bible, project_id=project_id),
   478	            contract_key="scene_bible",
   479	            agent_name="scene_bible",
   480	        )
   481	        # 重读 bible（人物谱可能已被并发流程更新），只覆盖 scenes 字段后回写。
   482	        p2 = conn.execute("SELECT bible_json FROM projects WHERE id=?", (project_id,)).fetchone()
   483	        data = json.loads(p2["bible_json"]) if p2 and p2["bible_json"] else bible.model_dump()
   484	        data["scenes"] = [s.model_dump() for s in scenes]
   485	        conn.execute(
   486	            "UPDATE projects SET bible_json=?,scene_refs_status='idle',scene_refs_error=NULL "
   487	            "WHERE id=?",
   488	            (json.dumps(data, ensure_ascii=False), project_id),
   489	        )
   490	        conn.commit()
   491	        recorder.succeed("场景设定已准备，场景图等待费用确认", conn=None)
   492	    except asyncio.CancelledError:
   493	        if task_registry.shutdown_in_progress():
   494	            recorder.pause_external("服务重启，场景设定任务等待自动恢复", conn=None)
   495	        else:
   496	            recorder.cancel(conn=None)
   497	        raise
   498	    except Exception as exc:  # noqa: BLE001 场景设定失败不阻断人物谱主流程
   499	        recorder.fail(exc, conn=None)
   500	        public = errors.record_and_format(exc, action="scene_bible_generate", context={"project_id": project_id})
   501	        conn.execute("UPDATE projects SET scene_refs_status='failed', scene_refs_error=? WHERE id=?",
   502	                     (public, project_id))
   503	        conn.commit()

   509	    与其在下次访问时判孤儿并报错，不如用持久化的 feedback 重新拉起任务续跑。"""
   510	    conn = get_conn()
   511	    style_column = "bible_style_name" if _supports_bible_style_name(conn) else "NULL AS bible_style_name"
   512	    rows = conn.execute(
   513	        f"SELECT id, bible_feedback, {style_column} FROM projects WHERE bible_status='running'"
   514	    ).fetchall()
   515	    resumed = 0
   516	    for r in rows:
   517	        pid = r["id"]
   518	        if _bible_task_active(pid):
   519	            continue
   520	        feedback = r["bible_feedback"] or ""
   521	        parent = conn.execute(
   522	            "SELECT id FROM workflow_runs WHERE workflow_type='character_bible' "
   523	            "AND scope_type='project' AND scope_id=? AND status='PAUSED_EXTERNAL' "
   524	            "AND recovered_by_run_id IS NULL ORDER BY updated_at DESC LIMIT 1",
   525	            (pid,),
   526	        ).fetchone()
   527	        recorder = None
   528	        try:
   529	            recorder = _new_bible_recorder(
   530	                pid, trigger_type="resume", requested_by="system",
   531	                parent_run_id=parent["id"] if parent else None,
   532	                style_name=r["bible_style_name"],
   533	            )
   534	            task_registry.spawn(
   535	                "bible",
   536	                pid,
   537	                _recorded_bible_task(
   538	                    pid, feedback, recorder, trigger_full_refs=True,
   539	                    style_name=r["bible_style_name"],
   540	                ),
   541	                project_id=pid,
   542	            )
   543	            resumed += 1
   544	        except Exception as exc:  # one project must not block all startup recovery
   545	            public = errors.record_and_format(
   546	                exc,
   547	                action="bible_recovery_spawn",
   548	                context={"project_id": pid},
   549	            )
   550	            conn.execute(
   551	                "UPDATE projects SET bible_status='failed', bible_error=? WHERE id=?",
   552	                (f"人物谱自动恢复未能启动，原文和反馈已保留，可重新发起。{public}", pid),
   553	            )
   554	            conn.commit()
   555	            if recorder is not None:
   556	                try:
   557	                    recorder.cancel("人物谱恢复任务未能启动", conn=None)
   558	                except Exception:  # noqa: BLE001
   559	                    pass
   560	    return resumed

   615	def recover_scene_ref_tasks() -> int:
   616	    """Resume persisted scene-asset work after a reload or process restart.
   617	
   618	    Scene generation is idempotent: approved references are skipped, so an
   619	    interrupted batch safely continues from the first missing scene instead of
   620	    regenerating accepted assets.
   621	    """
   622	    conn = get_conn()
   623	    rows = conn.execute(
   624	        "SELECT id,bible_json,bible_status,scene_refs_status,scene_refs_target "
   625	        "FROM projects WHERE scene_refs_status='running' "
   626	        "OR (scene_refs_status='idle' AND bible_status='ready')"
   627	    ).fetchall()
   628	    resumed = 0
   629	    for row in rows:
   630	        project_id = row["id"]
   631	        # A recovered character-bible task will start a fresh scene pipeline
   632	        # after committing its new Bible.  Starting from the old Bible here
   633	        # would race it and could generate obsolete assets.
   634	        if (row["bible_status"] == "running"
   635	                or _scene_assets_task_active(project_id)):
   636	            continue
   637	        try:
   638	            bible = json.loads(row["bible_json"] or "{}")
   639	        except (TypeError, ValueError, json.JSONDecodeError):
   640	            bible = {}
   641	        if bible.get("scenes"):
   642	            if row["scene_refs_status"] != "running":
   643	                continue
   644	            parent = conn.execute(
   645	                "SELECT id FROM workflow_runs WHERE workflow_type='scene_references' "
   646	                "AND scope_type='project' AND scope_id=? AND status='PAUSED_EXTERNAL' "
   647	                "AND recovered_by_run_id IS NULL ORDER BY updated_at DESC LIMIT 1",
   648	                (project_id,),
   649	            ).fetchone()
   650	            try:
   651	                if _start_scene_refs_generation(
   652	                    project_id,
   653	                    _decode_scene_target(row["scene_refs_target"]),
   654	                    resume=True,
   655	                    parent_run_id=parent["id"] if parent else None,
   656	                ):
   657	                    resumed += 1
   658	            except Exception as exc:  # noqa: BLE001
   659	                public = errors.record_and_format(
   660	                    exc, action="scene_refs_recovery_spawn",
   661	                    context={"project_id": project_id},
   662	                )
   663	                conn.execute(
   664	                    "UPDATE projects SET scene_refs_status='failed',scene_refs_error=? WHERE id=?",
   665	                    (f"场景图自动恢复未能启动，已完成素材仍保留，可重试缺口。{public}", project_id),
   666	                )
   667	                conn.commit()
   668	            continue
   669	        parent = conn.execute(
   670	            "SELECT id FROM workflow_runs WHERE workflow_type='scene_bible' "
   671	            "AND scope_type='project' AND scope_id=? AND status='PAUSED_EXTERNAL' "
   672	            "AND recovered_by_run_id IS NULL ORDER BY updated_at DESC LIMIT 1",
   673	            (project_id,),
   674	        ).fetchone()
   675	        try:
   676	            if _start_scene_bible_preparation(
   677	                project_id,
   678	                parent_run_id=parent["id"] if parent else None,
   679	                requested_by="system",
   680	                trigger_type="resume" if row["scene_refs_status"] == "running" else "automatic",
   681	            ):
   682	                resumed += 1
   683	        except Exception as exc:  # noqa: BLE001
   684	            public = errors.record_and_format(
   685	                exc, action="scene_bible_recovery_spawn", context={"project_id": project_id},
   686	            )
   687	            conn.execute(
   688	                "UPDATE projects SET scene_refs_status='failed',scene_refs_error=? WHERE id=?",
   689	                (f"场景清单自动恢复未能启动，可重新发起。{public}", project_id),
   690	            )
   691	            conn.commit()
   692	    return resumed

   694	async def _bible_task(
   695	    project_id: str,
   696	    feedback: str = "",
   697	    *,
   698	    trigger_full_refs: bool = True,
   699	    style_name: str | None = None,
   700	):
   701	    conn = get_conn()
   702	    try:
   703	        if style_name is None and _supports_bible_style_name(conn):
   704	            style_row = conn.execute(
   705	                "SELECT bible_style_name FROM projects WHERE id=?", (project_id,),
   706	            ).fetchone()
   707	            style_name = style_row["bible_style_name"] if style_row else None
   708	        style_name = _normalize_visual_style_name(style_name)
   709	        style_prompt = _visual_style_prompt_or_default(style_name)
   710	        chapters = rows_to_dicts(conn.execute(
   711	            "SELECT * FROM chapters WHERE project_id=? ORDER BY idx", (project_id,)).fetchall())
   712	        timeout_s = max(int(get_setting("bible_task_timeout_s") or BIBLE_TASK_TIMEOUT_S), 60)
   713	        # 重新谱写时按角色名保留已有定妆照（重生圣经不应丢失一致性锚点）
   714	        old_row = conn.execute("SELECT bible_json FROM projects WHERE id=?", (project_id,)).fetchone()
   715	        old_style = None
   716	        old_bible = None
   717	        if old_row and old_row["bible_json"]:
   718	            old_bible = json.loads(old_row["bible_json"])
   719	        from app import model_registry
   720	        from app.harness.text_provider_scope import stage_text_provider
   721	
   722	        resolved_text_provider = None
   723	        if "bible_text_provider" in _project_columns(conn):
   724	            provider_row = conn.execute(
   725	                "SELECT bible_text_provider FROM projects WHERE id=?", (project_id,),
   726	            ).fetchone()
   727	            resolved_text_provider = model_registry.resolve_stage_text_provider(
   728	                provider_row["bible_text_provider"] if provider_row else None
   729	            )
   730	        with stage_text_provider(resolved_text_provider):
   731	            bible = await asyncio.wait_for(
   732	                generate_bible(
   733	                    chapters, feedback=feedback, previous_bible=old_bible,
   734	                    project_id=project_id, visual_style_prompt=style_prompt,
   735	                ),
   736	                timeout=timeout_s,
   737	            )
   738	        if old_bible:
   739	            old_style = (old_bible.get("world") or {}).get("visual_style_canonical")
   740	            old_refs = {c.get("name"): c.get("ref_image_path")
   741	                        for c in old_bible.get("characters", [])}
   742	            for c in bible.characters:
   743	                c.ref_image_path = old_refs.get(c.name) or None
   744	        # 重谱后画风变化 → 旧画风定妆照与旧视频全部作废（否则图像信号会把新画风拉回旧画风）
   745	        if old_style and bible.world.visual_style_canonical != old_style:
   746	            _purge_for_style_change(project_id, bible)
   747	        residual = list(getattr(bible, "residual_errors", []) or [])
   748	        artifact_id = getattr(bible, "evidence_artifact_id", None)
   749	        bible_status = "warning" if residual else "ready"
   750	        bible_error = (
   751	            "人物谱存在阻塞问题，允许人工修订，但不会进入下游：" + "；".join(residual[:8])
   752	            if residual else None
   753	        )
   754	        # A few unit tests intentionally use a minimal legacy schema.  Production
   755	        # databases always receive the incremental migration in app.db, while the
   756	        # fallback keeps the stage function independently testable.
   757	        project_columns = {
   758	            row[1] for row in conn.execute("PRAGMA table_info(projects)").fetchall()
   759	        }
   760	        final_project_status = "bible_ready"
   761	        if "plan_status" in project_columns:
   762	            plan_row = conn.execute(
   763	                "SELECT plan_status FROM projects WHERE id=?", (project_id,)
   764	            ).fetchone()
   765	            if plan_row and plan_row["plan_status"] == "ready":
   766	                final_project_status = "planned"
   767	        if "bible_artifact_id" in project_columns:
   768	            conn.execute(
   769	                "UPDATE projects SET bible_json=?, bible_version=bible_version+1, bible_status=?, "
   770	                "bible_error=?, bible_artifact_id=?, status=? WHERE id=?",
   771	                (
   772	                    bible.model_dump_json(), bible_status, bible_error, artifact_id,
   773	                    final_project_status if not residual else "created", project_id,
   774	                ))
   775	        else:
   776	            conn.execute(
   777	                "UPDATE projects SET bible_json=?, bible_version=bible_version+1, bible_status=?, "
   778	                "bible_error=?, status=? WHERE id=?",
   779	                (
   780	                    bible.model_dump_json(), bible_status, bible_error,
   781	                    final_project_status if not residual else "created", project_id,
   782	                ))
   783	        conn.commit()
   784	        if trigger_full_refs and not residual:
   785	            try:
   786	                _start_refs_generation(project_id, None)
   787	            except Exception as exc:  # noqa: BLE001 bible remains deliverable
   788	                public = errors.record_and_format(
   789	                    exc, action="refs_spawn_after_bible", context={"project_id": project_id},
   790	                )
   791	                conn.execute(
   792	                    "UPDATE projects SET refs_status='failed',refs_error=? WHERE id=?",
   793	                    (f"人物谱已完成，但定妆任务未能启动，可直接重试定妆。{public}", project_id),
   794	                )
   795	                conn.commit()
   796	            if "scene_refs_status" in project_columns:
   797	                try:
   798	                    _start_scene_bible_preparation(project_id)
   799	                except Exception as exc:  # noqa: BLE001 bible remains deliverable
   800	                    public = errors.record_and_format(
   801	                        exc, action="scene_bible_spawn_after_bible",
   802	                        context={"project_id": project_id},
   803	                    )
   804	                    conn.execute(
   805	                        "UPDATE projects SET scene_refs_status='failed',scene_refs_error=? WHERE id=?",
   806	                        (f"人物谱已完成，但场景设定未能启动，可在场景库重试。{public}", project_id),
   807	                    )
   808	                    conn.commit()
   809	    except asyncio.TimeoutError:
   810	        conn.execute(
   811	            "UPDATE projects SET bible_status='failed', bible_error=? WHERE id=?",
   812	            (f"人物谱解析/修复超时（超过 {timeout_s} 秒），请重新谱写。", project_id),
   813	        )
   814	        conn.commit()
   815	    except asyncio.CancelledError:
   816	        if task_registry.shutdown_in_progress():
   817	            raise
   818	        row = conn.execute("SELECT bible_status FROM projects WHERE id=?", (project_id,)).fetchone()
   819	        if row and row["bible_status"] == "running":
   820	            conn.execute(
   821	                "UPDATE projects SET bible_status='failed', bible_error=? WHERE id=?",
   822	                (BIBLE_INTERRUPTED_ERROR, project_id),
   823	            )
   824	            conn.commit()
   825	        raise
   826	    except (StageError, Exception) as exc:  # noqa: BLE001
   827	        # 回滚必须在 errors.record_and_format() 之前（同一根因见
   828	        # app.domain.storyboard_ops._storyboard_task 顶层 except 上方的大注释：
   829	        # app.db.insert_error_log 在这同一个 task 缓存连接上落一条 error_logs 行
   830	        # 并 conn.commit()，谁先调用谁就先把此刻挂起的写入定型）。这条 try 里画风
   831	        # 变更时会走到 _purge_for_style_change → worker.purge_project_video_
   832	        # artifacts，后者对全项目逐镜头 DELETE shot_versions/shot_scenes/jobs、
   833	        # 逐集回退状态，整段过程故意不提交，只在处理完全部镜头后 commit 一次；
   834	        # 中途任何一步失败（文件 I/O、约束冲突等）都会把尚未提交的部分 DELETE
   835	        # 留在这个连接上。这里如果先记日志再回滚，日志落库的隐式 commit 会把这份
   836	        # 半成品（部分镜头的视频记录已删、其余镜头未处理）直接定型进库，且波及的
   837	        # 是整个项目而不止一集——这正是本文件同类 purge 调用（_refs_task 的
   838	        # errors.record_and_format 同理）必须先回滚的原因。回滚只丢弃这次失败尝试
   839	        # 自己产生的未提交写入，不影响更早已经各自 commit 过的检查点。
   840	        if conn.in_transaction:
   841	            conn.rollback()
   842	        public = errors.record_and_format(exc, action="bible_generate", context={"project_id": project_id})
   843	        conn.execute("UPDATE projects SET bible_status='failed', bible_error=? WHERE id=?", (public, project_id))
   844	        conn.commit()

   847	def _new_bible_recorder(
   848	    project_id: str,
   849	    *,
   850	    requested_by: str = "user",
   851	    trigger_type: str = "manual",
   852	    parent_run_id: str | None = None,
   853	    style_name: str | None = None,
   854	) -> WorkflowRecorder:
   855	    conn = get_conn()
   856	    chapters = rows_to_dicts(conn.execute(
   857	        "SELECT idx, title, content FROM chapters WHERE project_id=? ORDER BY idx", (project_id,)
   858	    ).fetchall())
   859	    if style_name is None and _supports_bible_style_name(conn):
   860	        row = conn.execute("SELECT bible_style_name FROM projects WHERE id=?", (project_id,)).fetchone()
   861	        style_name = row["bible_style_name"] if row else None
   862	    style_name = _normalize_visual_style_name(style_name)
   863	    project = conn.execute(
   864	        "SELECT bible_version, bible_feedback FROM projects WHERE id=?", (project_id,)
   865	    ).fetchone()
   866	    return WorkflowRecorder.create(
   867	        workflow_type="character_bible",
   868	        scope_type="project",
   869	        scope_id=project_id,
   870	        input_fingerprint=fingerprint(
   871	            chapters, project["bible_version"] if project else 0,
   872	            project["bible_feedback"] if project else None, style_name,
   873	        ),
   874	        requested_by=requested_by,
   875	        trigger_type=trigger_type,
   876	        policy_snapshot={"max_iterations": 4, "warning_blocks_downstream": True},
   877	        config_snapshot={"style_name": style_name},
   878	        parent_run_id=parent_run_id,
   879	    )

   882	async def _recorded_bible_task(
   883	    project_id: str,
   884	    feedback: str,
   885	    recorder: WorkflowRecorder,
   886	    *,
   887	    trigger_full_refs: bool,
   888	    style_name: str | None = None,
   889	) -> None:
   890	    recorder.start()
   891	    try:
   892	        conn = get_conn()
   893	        chapters = rows_to_dicts(conn.execute(
   894	            "SELECT idx, title, content FROM chapters WHERE project_id=? ORDER BY idx", (project_id,)
   895	        ).fetchall())
   896	        context = ContextPack(goal="生成可追溯人物圣经")
   897	        context.add_text("chapters", "\n\n".join(ch["content"] for ch in chapters), limit=60000)
   898	        await recorder.step(
   899	            "character_bible",
   900	            lambda: _bible_task(
   901	                project_id, feedback, trigger_full_refs=trigger_full_refs,
   902	                style_name=style_name,
   903	            ),
   904	            contract_key="character_bible",
   905	            agent_name="character_bible",
   906	            context_manifest=context.manifest(),
   907	        )
   908	        row = conn.execute("SELECT bible_status, bible_error FROM projects WHERE id=?", (project_id,)).fetchone()
   909	        if row and row["bible_status"] == "ready":
   910	            recorder.succeed("人物谱已通过确定性门禁", conn=None)
   911	        elif row and row["bible_status"] == "warning":
   912	            recorder.partial(row["bible_error"] or "人物谱需要人工修订", conn=None)
   913	        else:
   914	            recorder.fail(RuntimeError(row["bible_error"] if row else "人物谱生成失败"), conn=None)
   915	    except asyncio.CancelledError:
   916	        if task_registry.shutdown_in_progress():
   917	            recorder.pause_external("服务重启，人物谱任务等待自动恢复", conn=None)
   918	        else:
   919	            recorder.cancel(conn=None)
   920	        raise
   921	    except Exception as exc:
   922	        recorder.fail(exc, conn=None)
   923	        raise

   926	async def _start_bible_core(
   927	    project_id: str,
   928	    feedback: str,
   929	    *,
   930	    confirm: bool = False,
   931	    quote_id: str | None = None,
   932	    require_quote_id: bool = False,
   933	    style_name: str | None = None,
   934	) -> dict:
   935	    """启动人物谱生成的领域逻辑，供 REST 路由与 ``bible.generate`` Command Handler 共用。"""
   936	    p = _project_or_404(project_id)
   937	    _require_harness_engine(project_id)
   938	    if p["bible_status"] == "running" and _bible_task_active(project_id):
   939	        raise HTTPException(409, "角色圣经正在生成中")
   940	    if p["refs_status"] == "running":
   941	        raise HTTPException(409, "定妆照正在生成中，请先停止后再重生人物谱")
   942	    feedback = feedback.strip()
   943	    if len(feedback) > 2000:
   944	        raise HTTPException(400, "打回要求过长，请控制在 2000 字以内")
   945	    style_name = _normalize_visual_style_name(style_name)
   946	    precheck = _compute_bible_generate_precheck(project_id, style_name=style_name)
   947	    if not confirm:
   948	        raise _payment_confirm_required(precheck)
   949	    quote_row = _validate_payment_quote(project_id, quote_id, precheck)
   950	    if quote_row["consumed_at"] is not None:
   951	        return {
   952	            "status": "accepted", "idempotent_replay": True,
   953	            "task_id": quote_row["consumed_task_id"], "run_id": quote_row["consumed_run_id"],

  1142	def _classify_bible_changes(old_bible: dict | None, new_bible: dict) -> list[str]:
  1143	    """区分仅文字 / 角色外观 / 全局画风变更，供定稿影响预检展示。"""
  1144	    changes: list[str] = []
  1145	    old = old_bible or {}

  1147	    new_style = (new_bible.get("world") or {}).get("visual_style_canonical")
  1148	    if old_style and new_style and old_style != new_style:
  1149	        changes.append("global_style")
  1150	    old_chars = {c.get("name"): c for c in old.get("characters", []) if c.get("name")}
  1151	    new_chars = {c.get("name"): c for c in new_bible.get("characters", []) if c.get("name")}
  1152	    appearance_changed = False
  1153	    text_changed = False
  1154	    if set(old_chars) != set(new_chars):
  1155	        text_changed = True
  1156	    for name, nc in new_chars.items():
  1157	        oc = old_chars.get(name) or {}
  1158	        if (oc.get("appearance_canonical") or "") != (nc.get("appearance_canonical") or ""):
  1159	            appearance_changed = True
  1160	        for field in ("personality", "speech_style", "role", "portrait_prompt_override"):
  1161	            if (oc.get(field) or "") != (nc.get(field) or ""):
  1162	                text_changed = True
  1163	        if (oc.get("relationships") or []) != (nc.get("relationships") or []):
  1164	            text_changed = True
  1165	    if appearance_changed:
  1166	        changes.append("character_appearance")
  1167	    if text_changed and "character_appearance" not in changes:
  1168	        changes.append("text_only")
  1169	    elif text_changed:
  1170	        changes.append("text_fields")
  1171	    if not changes:
  1172	        changes.append("text_only")
  1173	    return changes
