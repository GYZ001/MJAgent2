  4045	    证据），但"区间边界没独立证据"与"核心事实没证据"是两件不同的事，不能绑在一起

  4144	    半角引号导致的假阴性）下同时命中，不能分别用不同形式各自命中一侧再拼凑。任一侧

  4149	    候选判别）与后续段号钉证，只是额外补上"引句本身双锚定"这一层，四闸独立生效。

  4157	        return False

  4159	        if (
  4160	            any(s in candidate for s in subject_forms)
  4161	            and any(o in candidate for o in object_forms)
  4162	        ):
  4163	            return True
  4164	    return False

  4541	def _bible_paratext_scope(valid: list[dict]) -> list[int]:
  4542	    """人物谱真正会读到的章（下标落在 `valid` 上）：头部 + 后段抽样 + 必收统计窗口。
  4543	
  4544	    净化按章一次模型调用，范围必须跟「读了什么」对齐，否则代价随书长线性增长
  4545	    而收益为零。
  4546	    """
  4547	    plan = _bible_source_plan(valid, BIBLE_SOURCE_BUDGET_CHARS, BIBLE_HEAD_CHAPTERS)
  4548	    scope = {index for index, _, _ in plan}
  4549	    head_end = max((index for index, _, excerpt in plan if not excerpt), default=-1) + 1
  4550	    # 净化只会让正文变短，头部因此可能比按原文规划时多吃进几章；同时
  4551	    # `_recurring_character_names` 的逐字统计窗口是前 HEAD+LOOKAHEAD 章，
  4552	    # 必收名单正是被旁文本污染的那一环，这段必须整段净化。
  4553	    window = max(
  4554	        head_end + BIBLE_PARATEXT_MARGIN_CHAPTERS,
  4555	        BIBLE_HEAD_CHAPTERS + BIBLE_LOOKAHEAD_CHAPTERS,
  4556	    )
  4557	    scope |= set(range(min(window, len(valid))))
  4558	    return sorted(index for index in scope if 0 <= index < len(valid))

  4688	def _character_detail_evidence_pack(
  4689	    chapters: list[dict], appellations: list[str], *, max_chars: int = BIBLE_DETAIL_EVIDENCE_MAX_CHARS,
  4690	) -> str:
  4691	    """Deterministically retrieve a bounded source dossier for one character."""
  4692	    anchors = [value.strip() for value in appellations if value and value.strip()]
  4693	    selected: list[str] = []
  4694	    seen: set[tuple[int, str]] = set()
  4695	    for chapter in chapters:
  4696	        content = (chapter.get("content") or "").strip()
  4697	        if not content:
  4698	            continue
  4699	        try:
  4700	            chapter_idx = int(chapter.get("idx"))
  4701	        except (TypeError, ValueError):
  4702	            continue
  4703	        for segment in index_source_segments(content, max_chars=800):
  4704	            text = segment.text.strip()
  4705	            if not text or not any(anchor in text for anchor in anchors):
  4706	                continue
  4707	            key = (chapter_idx, text)
  4708	            if key in seen:
  4709	                continue
  4710	            block = f"【第{chapter_idx}章·证据】\n{text}"
  4711	            if sum(len(item) for item in selected) + len(block) > max_chars:
  4712	                return "\n\n".join(selected)
  4713	            selected.append(block)
  4714	            seen.add(key)
  4715	            if len(selected) >= BIBLE_DETAIL_EVIDENCE_MAX_SEGMENTS:
  4716	                return "\n\n".join(selected)
  4717	    if selected:
  4718	        return "\n\n".join(selected)
  4719	    # No lexical hit: bounded fallback only, never the 60K source corpus.
  4720	    for chapter in chapters[:3]:
  4721	        content = (chapter.get("content") or "").strip()
  4722	        if not content:
  4723	            continue
  4724	        block = f"【第{chapter.get('idx', '?')}章·有限背景】\n{content[:1200]}"
  4725	        if sum(len(item) for item in selected) + len(block) > max_chars:
  4726	            break
  4727	        selected.append(block)
  4728	    return "\n\n".join(selected)

  5060	def _source_dialogue_evidence(text: str, limit: int) -> str:
  5061	    """Select exact dialogue evidence from a source section without inventing prose.
  5062	
  5063	    Long chapters used to be truncated at the head, which made dialogue in the
  5064	    middle or climax literally unavailable to generation and repair.  Preserve
  5065	    dialogue-shaped source fragments in addition to stable head/tail context.
  5066	    """
  5067	    if limit <= 0 or not text:
  5068	        return ""
  5069	    fragments: list[str] = []
  5070	    seen: set[str] = set()
  5071	    for pattern in (_SOURCE_SPEAKER_DIALOGUE_RE, _SOURCE_QUOTED_DIALOGUE_RE):
  5072	        for match in pattern.finditer(text):
  5073	            fragment = match.group(0).strip()
  5074	            condensed = re.sub(r"\s+", "", fragment)
  5075	            if not condensed or condensed in seen:
  5076	                continue
  5077	            seen.add(condensed)
  5078	            fragments.append(fragment)
  5079	    selected: list[str] = []
  5080	    used = 0
  5081	    for fragment in fragments:
  5082	        cost = len(fragment) + (1 if selected else 0)
  5083	        if used + cost > limit:
  5084	            continue
  5085	        selected.append(fragment)
  5086	        used += cost
  5087	    return "\n".join(selected)

  5090	def _render_screenplay_source(source_text: str, budget: int = SCREENPLAY_SOURCE_BUDGET_CHARS) -> str:
  5091	    text = source_text or ""
  5092	    if len(text) <= budget:
  5093	        return text
  5094	    marker_a = "\n\n……（中段叙事已按上下文预算压缩；以下保留中段原文对白证据）……\n"
  5095	    marker_b = "\n\n……（继续保留本章结尾原文，结尾事件与台词不得遗漏）……\n"
  5096	    payload_budget = max(0, budget - len(marker_a) - len(marker_b))
  5097	    head_budget = int(payload_budget * 0.35)
  5098	    tail_budget = int(payload_budget * 0.35)
  5099	    dialogue_budget = payload_budget - head_budget - tail_budget
  5100	    middle_end = max(head_budget, len(text) - tail_budget)
  5101	    middle = text[head_budget:middle_end]
  5102	    dialogue_evidence = _source_dialogue_evidence(middle, dialogue_budget)
  5103	    # If the middle contains little dialogue, use the remaining allowance for
  5104	    # contiguous context immediately after the head instead of wasting budget.
  5105	    unused = max(0, dialogue_budget - len(dialogue_evidence))
  5106	    head = text[:head_budget + unused]
  5107	    tail = text[-tail_budget:] if tail_budget else ""
  5108	    return head + marker_a + dialogue_evidence + marker_b + tail
