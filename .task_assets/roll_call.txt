  1941	def _bible_source_plan(
  1942	    valid: list[dict], budget: int, head_chapters: int | None,
  1943	) -> list[tuple[int, int, bool]]:
  1944	    """规划人物谱源文本读哪几章、每章读多少字：`(章在 valid 里的下标, 截取字数, 是否节选)`。
  1945	
  1946	    渲染（`_render_bible_source`）和「哪几章需要先净化旁文本」
  1947	    （`_bible_paratext_scope`）共用这一份规划：口径只有一处，不会漂移出
  1948	    「净化了 643 章、真正读的只有 33 章」那种落差。
  1949	    """
  1950	    plan: list[tuple[int, int, bool]] = []
  1951	    # 头部顺序铺设：用至多 70% 预算（其余留给后段抽样）。
  1952	    head_budget = int(budget * 0.7)
  1953	    if head_chapters:
  1954	        # 首版人物谱要求「完整读完前 N 章」。按比例切的头部会随章节长度漂移：
  1955	        # 长章小说可能读到第三、四章就把头部预算用光，主要配角整体缺席。
  1956	        head_budget = min(budget, max(head_budget, sum(
  1957	            len(ch["content"].strip()) for ch in valid[:head_chapters]
  1958	        )))
  1959	    used = 0
  1960	    head_count = 0
  1961	    for index, ch in enumerate(valid):
  1962	        remain = head_budget - used
  1963	        if remain <= 200:
  1964	            break
  1965	        take = min(len(ch["content"].strip()), remain)
  1966	        plan.append((index, take, False))
  1967	        used += take
  1968	        head_count += 1
  1969	
  1970	    # 后段抽样：在头部未覆盖的章节里均匀取样，注入每章开头若干字，覆盖后期登场人物。
  1971	    later = valid[head_count:]
  1972	    remain_budget = budget - used
  1973	    if later and remain_budget > 200:
  1974	        sample_n = min(len(later), _BIBLE_TAIL_SAMPLE_MAX, max(1, remain_budget // _BIBLE_TAIL_SLICE_CHARS))
  1975	        if sample_n > 0:
  1976	            step = len(later) / sample_n
  1977	            picked_idx = sorted({min(len(later) - 1, int(i * step)) for i in range(sample_n)})
  1978	            for li in picked_idx:
  1979	                if remain_budget <= 200:
  1980	                    break
  1981	                content = later[li]["content"].strip()
  1982	                take = min(len(content), _BIBLE_TAIL_SLICE_CHARS, remain_budget)
  1983	                plan.append((head_count + li, take, True))
  1984	                remain_budget -= take
  1985	    return plan

  1988	def _render_bible_source(chapters: list[dict], budget: int = BIBLE_SOURCE_BUDGET_CHARS,
  1989	                         *, head_chapters: int | None = None) -> str:
  1990	    """为角色圣经渲染源文本：先顺序铺头部（主角通常在前期出场），再在剩余预算里
  1991	    跨越全书【抽样后段章节的开头】，让后期才登场的重要角色（如中后段反派）也能进圣经——
  1992	    否则分镜阶段引用这些角色会因"不在圣经"而反复返工或被迫漏掉。
  1993	    """
  1994	    valid = [ch for ch in chapters if (ch.get("content") or "").strip()]
  1995	    if not valid:
  1996	        return ""
  1997	
  1998	    def _title(ch: dict) -> str:
  1999	        return ch.get("title") or f"第{ch.get('idx', '?')}章"
  2000	
  2001	    blocks: list[str] = []
  2002	    for index, take, excerpt in _bible_source_plan(valid, budget, head_chapters):
  2003	        ch = valid[index]
  2004	        content = ch["content"].strip()
  2005	        clipped = content[:take]
  2006	        if excerpt:
  2007	            suffix = "……（节选开头，仅供识别后期登场角色）" if len(content) > take else ""
  2008	            blocks.append(f"【{_title(ch)}·节选】\n{clipped}{suffix}")
  2009	        else:
  2010	            suffix = "……（原文过长已截断）" if len(content) > take else ""
  2011	            blocks.append(f"【{_title(ch)}】\n{clipped}{suffix}")
  2012	
  2013	    return "\n\n".join(blocks)

  2044	async def _recurring_character_names(
  2045	    chapters: list[dict], *, project_id: str | None = None,
  2046	) -> list[tuple[str, str, int]]:
  2047	    """产出「必收角色名单」：先点名+自报在场证据，再用结构闸+独立裁决闸核验每条
  2048	    证据是不是真的证明本人在场，核验通过的证据条数（`verified_onstage_count`）
  2049	    才是判据——不再是"名字字符串在原文窗口里出现的次数"。
  2050	
  2051	    根因：旧判据把字符串出现次数当"重不重要"的代理信号，两个方向都会失效。
  2052	    假阳性方向——王伯/周员外/靠山老祖的命中全部来自旁白交代身份或他人台词提及，
  2053	    本人从未真正在场，却因为次数够多进了必收名单。假阴性方向——这个信号只统计
  2054	    模型报出的候选名字本身的出现次数，原文如果通篇用绰号称呼一个人（本人几乎
  2055	    只以"小胖子"出现，正式姓名仅出现一两次）而此刻圣经正文还没生成、没有别名表
  2056	    能把绰号翻译回正式姓名，这个人就会被判定为不重要。
  2057	
  2058	    新流程：
  2059	    1. 模型只在前 BIBLE_HEAD_CHAPTERS 章里点名，每个候选申报 primary_appellation
  2060	       （原文最常用写法，允许绰号）+ formal_name（原文已揭示的正式姓名，未揭示则
  2061	       空）+ onstage_evidence（能证明本人在场的原文引句列表）——绰号本身就能直接
  2062	       充当"必收货币"，不需要一张此刻还不存在的别名表做转译。
  2063	    2. 代码结构闸，逐条证据：G1 引句所在章节必须落在统计窗口（前 HEAD+LOOKAHEAD
  2064	       章）内；G2 引句必须逐字命中该章原文（允许模型自行加/脱一层引号的噪音）；
  2065	       G3 称呼（primary_appellation 或 formal_name 中非空的那个）必须是引句子串。
  2066	       任一不满足直接丢弃该条证据，不发起裁决调用。
  2067	    3. 结构闸通过的证据才发起独立低温模型裁决（`_roster_presence_verdict_call`，
  2068	       与别名裁决闸同一分工范式：代码检索卷宗 → 模型独立裁决 → 代码结构性钉证）：
  2069	       只问这段原文里称呼所指的人物本人是不是真的在场（本人说话/动作/被叙述在场，
  2070	       而不是被谈论、被指涉、被交代来历的对象）；裁决通过（verdict=="onstage" 且
  2071	       段号钉证通过）才计入该候选的 `verified_onstage_count`。
  2072	    4. 按 `verified_onstage_count` 降序（同分按 primary_appellation 字典序打破
  2073	       平局）排序，取 >= BIBLE_RECURRING_MIN_ONSTAGE_QUOTES 的候选，最多保留
  2074	       BIBLE_MUST_COVER_MAX 个。
  2075	
  2076	    点名调用失败时返回空名单，绝不阻断人物谱本身；结构闸/裁决闸任一步不通过，
  2077	    该条证据直接丢弃，不确定不登记（不会因为某一条证据没通过就拒绝整个候选，
  2078	    只是那一条不计数）。
  2079	    """
  2080	    valid = [ch for ch in chapters if (ch.get("content") or "").strip()]
  2081	    if not valid:
  2082	        return []
  2083	    head = valid[:BIBLE_HEAD_CHAPTERS]
  2084	    head_text = _render_bible_source(head, head_chapters=BIBLE_HEAD_CHAPTERS)
  2085	    if not head_text.strip():
  2086	        return []

  2088	
  2089	要求：
  2090	1. primary_appellation：原文里称呼这个人物最常用、最稳定的一种写法——可以是正式姓名，也可以是外号、
  2091	   绰号、尊称或代称（如"小胖子""许师姐""靠山老祖"），取原文实际出现频率最高的那一种写法，逐字照抄，
