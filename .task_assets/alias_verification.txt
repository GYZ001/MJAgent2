  3532	        form for name in candidates for form in roster.get(name, [])
  3533	    }

  3535	        resolved_chapter_index, chapter_text, text, dossier_anchor_texts,
  3536	    )
  3537	    if not dossier:
  3538	        return {**empty, "reason": "no_verdict_dossier"}

  3540	        build_chapter_cognition_card(
  3541	            bible, chapters_by_idx, resolved_chapter_index, character_names=candidates,
  3542	        )
  3543	        if bible is not None else None
  3544	    )

  3546	        response = await _alias_verdict_call(
  3547	            alias=text, true_name=true_name, dossier=dossier,
  3548	            candidates=candidates, project_id=project_id,
  3549	            cognition_card=cognition_card,

  3551	    except Exception as exc:  # noqa: BLE001 - 裁决调用失败按不确定处理：不确定不登记
  3552	        log_provider_call(
  3553	            "character_alias_backfill_verdict", config.MODEL_TEXT, "FAILED", None, 0,

  3555	        )

  3559	            "candidate_uncertain"
  3560	            if response.selected_candidate == _ALIAS_VERDICT_NO_MATCH_LABEL
  3561	            else "candidate_mismatch"
  3562	        )
  3563	        return {**empty, "reason": reason}
  3564	    if _alias_verdict_pin_segment(dossier, response.supporting_segment_index) is None:
  3565	        return {**empty, "reason": "segment_not_pinned"}

  3567	        "accepted": True, "chapter_idx": resolved_chapter_index,

  3569	    }

  3630	def _render_alias_backfill_source(
  3631	    chapters: list[dict], budget: int = ALIAS_BACKFILL_SOURCE_BUDGET_CHARS,
  3632	) -> str:
  3633	    """为别名回填渲染全书原文：块头强制显示原文章节序号（idx），不像 `_render_bible_source`
  3634	    那样优先用章节标题——回填要求模型精确报出 `evidence_chapter_index`，标题文本
  3635	    （可能是任意小说章节名）无法保证与 idx 对应，块头必须显式给出数字。"""
  3636	    valid = [ch for ch in chapters if (ch.get("content") or "").strip()]
  3637	    blocks: list[str] = []
  3638	    used = 0
  3639	    for chapter in valid:
  3640	        remain = budget - used
  3641	        if remain <= 200:
  3642	            break
  3643	        content = chapter["content"].strip()
  3644	        clipped = content[:remain]
  3645	        suffix = "……（原文过长已截断）" if len(content) > remain else ""
  3646	        blocks.append(f"【第 {chapter.get('idx', '?')} 章】\n{clipped}{suffix}")
  3647	        used += len(clipped)
  3648	    return "\n\n".join(blocks)
